## 3. Abordagem de Implementação

### 3.1 Criar Conversor

**Arquivo:** `/services/optimizer-agents/src/services/hypothesis_converter.py`

```python
from services.optimizer_agents.src.models.optimization_hypothesis import OptimizationHypothesis
from hypothesis_library_client import HypothesisLibraryClient

class HypothesisConverter:
    """Converte OptimizationHypothesis para HypothesisCreate"""
    
    async def to_hypothesis_create(self, opt_hypothesis: OptimizationHypothesis) -> dict:
        """Converte para formato do hypothesis-library"""
        return {
            "title": f"Hypothesis: {opt_hypothesis.id}",
            "description": opt_hypothesis.description,
            "background": f"Generated by optimizer: {opt_hypothesis.optimizer_type}",
            "expected_outcome": opt_hypothesis.expected_improvement,
            "metrics": opt_hypothesis.target_metrics,
            "baseline_metrics": opt_hypothesis.current_metrics,
            "priority": self._map_priority(opt_hypothesis.priority),
            "author": "optimizer-agents",
            "requires_experiment": True,
            "metadata": {
                "optimizer_hypothesis_id": str(opt_hypothesis.id),
                "optimizer_type": opt_hypothesis.optimizer_type
            }
        }
    
    def _map_priority(self, priority: str) -> str:
        """Mapeia prioridade do optimizer para hypothesis-library"""
        mapping = {"critical": "CRITICAL", "high": "HIGH", "medium": "MEDIUM"}
        return mapping.get(priority.lower(), "LOW")
```

### 3.2 Adicionar Cliente HTTP no Optimizer

**Arquivo:** `/services/optimizer-agents/src/clients/hypothesis_library_client.py`

```python
import httpx
from typing import Dict, Any

class HypothesisLibraryClient:
    """Cliente HTTP para hypothesis-library"""
    
    def __init__(self, base_url: str = "http://hypothesis-library:8001"):
        self.base_url = base_url
        self.client = httpx.AsyncClient(timeout=30.0)
    
    async def create_hypothesis(self, hypothesis_data: dict) -> Dict[str, Any]:
        """Cria hipótese no hypothesis-library"""
        response = await self.client.post(
            f"{self.base_url}/api/v1/hypotheses",
            json=hypothesis_data
        )
        response.raise_for_status()
        return response.json()
    
    async def update_hypothesis_results(self, hypothesis_id: str, results: dict) -> Dict[str, Any]:
        """Atualiza hipótese com resultados de experimento"""
        response = await self.client.post(
            f"{self.base_url}/api/v1/hypotheses/{hypothesis_id}/complete",
            json={"results": results}
        )
        response.raise_for_status()
        return response.json()
```

### 3.3 Modificar ExperimentManager

**Arquivo:** `/services/optimizer-agents/src/services/experiment_manager.py`

```python
from .hypothesis_converter import HypothesisConverter
from .clients.hypothesis_library_client import HypothesisLibraryClient

class ExperimentManager:
    def __init__(self, ...):
        self.hypothesis_converter = HypothesisConverter()
        self.hypothesis_client = HypothesisLibraryClient()
    
    async def submit_experiment_with_hypothesis(self, opt_hypothesis: OptimizationHypothesis):
        """Submete experimento e salva hipótese"""
        # 1. Converter para formato do hypothesis-library
        hypothesis_data = await self.hypothesis_converter.to_hypothesis_create(opt_hypothesis)
        
        # 2. Salvar no hypothesis-library
        created_hypothesis = await self.hypothesis_client.create_hypothesis(hypothesis_data)
        hypothesis_id = created_hypothesis["hypothesis_id"]
        
        # 3. Submeter experimento
        experiment = await self.submit_experiment(opt_hypothesis, hypothesis_id)
        
        # 4. Atualizar hipótese com experiment_id
        await self.hypothesis_client.start_test(hypothesis_id, experiment["experiment_id"])
        
        return experiment
```
